빠른 시작

🚀 서버 실행

# 저장소 클론
git clone <%= githubUrl %>.git
cd APIServer

# 의존성 설치
npm install

# 개발 서버 실행
npm run dev

# 프로덕션 서버 실행
npm start

📖 API 사용 예제

// Todo 목록 조회
fetch('<%= apiUrl %>/todo')
  .then(response => response.json())
  .then(data => console.log(data));

// 새 할일 생성
fetch('<%= apiUrl %>/todo', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    title: '새로운 할일',
    content: '할일 내용'
  })
});
